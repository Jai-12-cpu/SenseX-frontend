import React, { useEffect, useState, useRef } from 'react'; import { StyleSheet, View, Text, Vibration, PanResponder } from 'react-native'; import * as Speech from 'expo-speech'; const WS_URL = "wss://sensex-backend.onrender.com"; export default function App() { const [status, setStatus] = useState("Connecting..."); const ws = useRef(null); useEffect(() => { connectWebSocket(); return () => ws.current?.close(); }, []); const connectWebSocket = () => { ws.current = new WebSocket(WS_URL); ws.current.onopen = () => setStatus("Connected"); ws.current.onmessage = (e) => { const data = JSON.parse(e.data); if (data.type === "VIBRATE") { // Trigger physical motor based on backend pattern Vibration.vibrate(data.pattern); } }; ws.current.onclose = () => { setStatus("Disconnected. Reconnecting..."); setTimeout(connectWebSocket, 3000); // Auto-reconnect }; }; // Gesture Recognition: Swipe up/down/left/right to "speak" const panResponder = PanResponder.create({ onStartShouldSetPanResponder: () => true, onPanResponderRelease: (evt, gestureState) => { if (gestureState.dy < -100) handleAction("I need help", [500]); else if (gestureState.dx > 100) handleAction("Yes", [100]); else if (gestureState.dx < -100) handleAction("No", [300]); }, }); const handleAction = (message, confirmVibration) => { // 1. Text-to-Speech for others to hear Speech.speak(message); // 2. Haptic confirmation for the user Vibration.vibrate(confirmVibration); // 3. Inform backend via WebSocket ws.current?.send(JSON.stringify({ action: message })); }; return ( {status} Swipe UP for Help Swipe RIGHT for Yes Swipe LEFT for No ); } const styles = StyleSheet.create({ container: { flex: 1, backgroundColor: '#000', justifyContent: 'center', alignItems: 'center' }, indicator: { position: 'absolute', top: 50, padding: 10, borderRadius: 5, backgroundColor: '#222' }, statusText: { color: '#0f0', fontWeight: 'bold' }, instruction: { color: '#444', fontSize: 20, marginVertical: 10 } });
